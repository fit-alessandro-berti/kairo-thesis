\def\KairoClusterFrame{%
  \path[use as bounding box] (0,0) rectangle (4.24,2.24);
  \fill[KairoPaper] (0.12,0.61) rectangle (4.12,2.17);}
\def\KairoClusterGrid{%
  \draw[KairoLine!46,line width=0.35pt]
    (0.12,1.13) -- (4.12,1.13)
    (0.12,1.65) -- (4.12,1.65)
    (1.12,0.61) -- (1.12,2.17)
    (2.12,0.61) -- (2.12,2.17)
    (3.12,0.61) -- (3.12,2.17);
  \draw[KairoLine,line width=0.45pt] (0.12,0.61) rectangle (4.12,2.17);}
\def\KairoClusterPoints#1#2#3{%
  \foreach \px/\py in #3
    \draw[draw=#1,fill=#2,line width=0.52pt] (\px,\py) circle[radius=0.048];}
\def\KairoCentroid#1#2#3{%
  \begin{scope}[shift={(#2,#3)}]
    \draw[draw=#1,fill=white,line width=0.65pt] (0,0) circle[radius=0.112];
    \draw[#1,line width=0.65pt] (-0.060,0) -- (0.060,0) (0,-0.060) -- (0,0.060);
  \end{scope}}
\def\KairoClusterCross#1#2{%
  \begin{scope}[shift={(#1,#2)}]
    \draw[KairoInk,line width=0.72pt] (-0.066,-0.066) -- (0.066,0.066)
      (-0.066,0.066) -- (0.066,-0.066);
  \end{scope}}

}
